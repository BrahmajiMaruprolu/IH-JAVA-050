package Mypac;

import java.io.*;
import javax.servlet.*;
import javax.servlet.http.*;


public class MainServlet extends HttpServlet {

    @Override
    protected void service(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
         resp.setContentType("text/html");
         PrintWriter pw=resp.getWriter();
         String sbook=req.getParameter("book");
         Cookie c=new Cookie("bookname",sbook);
         resp.addCookie(c);
         ServletContext sc=getServletContext();
         sc.setAttribute("bname",sbook);
         resp.sendRedirect("CookieServlet");
    }

}

import javax.servlet.http.*;


public class CookieServlet extends HttpServlet {

    @Override
    protected void service(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
         resp.setContentType("text/html");
         PrintWriter pw=resp.getWriter();
         Cookie mycks[]=req.getCookies();
         String bname=null;
         ServletContext sc=getServletContext();
         bname=(String)sc.getAttribute("bname");
         for(Cookie c:mycks){
             if(c.getName().equals("bname")){
                bname=c.getValue();
            }
            
         }
          RequestDispatcher rd=req.getRequestDispatcher(bname+".html");
             rd.include(req, resp);
         pw.print("<h1 align=center style=color:green>Select more than one book to get 25% discount</h1>");
    }

}

package Mypac;

import java.io.*;
import javax.servlet.*;
import javax.servlet.http.*;


public class AddToCartServlet extends HttpServlet {

    @Override
    protected void service(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
       resp.setContentType("text/html");
       PrintWriter pw=resp.getWriter();
       ServletContext sc=getServletContext();
       int bcount=0,bprice=0,bfinal_price=0;
       String btitle;
        String books[]=req.getParameterValues("cb");
        String bt[]=new String[books.length];
           int bp[]=new int[books.length];
           pw.print("<div>");
           pw.print("<body style= background-color:black>");
           pw.print("<h2 style=color:red align=center>Here are your Cart Details Please Check Once And Proceed</h2>");
           pw.print("<table border=1 align=center style= background-color:mediumspringgreen text-align:center>");
           pw.print("<tr>");
           pw.print("<th>Sno</th>");
           pw.print("<th>Book Name</th>");
           pw.print("<th>Book Price</th>");
           pw.print("</tr>");
      try{ if(req.getParameterValues("cb")!=null){
           for(int i=0;i<books.length;i++){
               int len=Integer.parseInt(books[i].substring(0,2));
               btitle=books[i].substring(2,len-1);
               bt[i]=btitle;
               bprice=Integer.parseInt(books[i].substring(len,books[i].length()));
               bp[i]=bprice;
           }
       }
       else{
           String bname="";
           RequestDispatcher rd=req.getRequestDispatcher((String) sc.getAttribute(bname+".html"));
           rd.include(req, resp);
       }}catch(Exception e){
           System.out.println("Exception ");
       }
       for(int i=0;i<books.length;i++){
           
           pw.print("<tr>");
           pw.print("<th>"+(i+1)+"</th>");
           pw.print("<th>"+bt[i]+"</th>");
           pw.print("<th>"+bp[i]+"</th>");
           pw.print("</tr>");
          
           bfinal_price=bfinal_price+bp[i];
           bcount=bcount+1;
       }
       pw.print("<tr>");
       pw.print("<th>");
        pw.print("<button><a href=index.html>back</a></button>");
        pw.print("</th>");
       
        
        pw.print("<th>");
        pw.print("---Cart Details---");
         pw.print("</th>");
         
        
        pw.print("<th>");
        pw.print("<button><a href=PaymentGateway.html>proceed</a></button>");
         pw.print("</th>");
        pw.print("</tr>");
        pw.print("</table>");
        pw.print("</body>");
        pw.print("</div>");
        if(bcount>1){
            bfinal_price=(int)(bfinal_price-(bfinal_price*0.25));
        }
        else{
            pw.print("<h1 align=center style=color:red>You Missed The Offer</h1>");
        }
        pw.print("<h1 align=center style=color:hotpink>Cart Amount:"+bfinal_price+"</h1>");
        sc.setAttribute("final_amount",bfinal_price);
    }

    
}

package Mypac;

import java.io.*;
import javax.servlet.*;
import javax.servlet.http.*;
import java.sql.*;
import java.util.Random;
public class PaymentServlet extends HttpServlet {
    ResultSet rs;
    Statement stmt;
      Connection con;
    @Override
    public void init() throws ServletException {
        try{
         Class.forName("org.apache.derby.jdbc.ClientDriver");
        con=DriverManager.getConnection("jdbc:derby://localhost:1527/cd");
         }catch(Exception e){
                 System.out.println("Exception Handled");
                 }
    }
    

    @Override
    protected void service(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
         resp.setContentType("text/html");
         PrintWriter pw=resp.getWriter();
         ServletContext sc=getServletContext();
         String name,cvv,psd,s1,cno;
          int price;
         HttpSession hs=req.getSession(false);
         if(hs==null){
             pw.print("<h1 align=center style=color:red>Time out..Please try another time</h1>");
             pw.print("<h1 align=center style=color:blue>Click the below link to go to home page</h1>");
             pw.print("<h1 align=center style=color:green><button><a href=index.html>back</a></button></h1>");
         }
         else{  
             name=(String) sc.getAttribute("name");
             cvv=(String) sc.getAttribute("cvv");
             cno= (String) sc.getAttribute("cno");
             psd=(String) sc.getAttribute("psd");
             price=(int) sc.getAttribute("final_amount");
             String s="select * from cart";
             try{ 
                 stmt=con.createStatement();
                 rs=stmt.executeQuery(s);
                 if(rs!=null){
                        while(rs.next()){
                            if(rs.getString("cno").equals(cno)){
                                if(rs.getString("cvv").equals(cvv) && rs.getString("psd").equals(psd)){

                                        //pw.print(rs.getInt(5));
                                        //pw.print(price);
